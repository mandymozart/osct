  master?: number;
  slots: Record<string, SoundSlot>;

interface SoundSlot {
  preset: string | SoundPreset;
  listens?: (string | { event; where? })[];
  volume?; muted?; loop?; cooldown?;
  haptic?: number | number[];
  replaces?: string;      // target: instead of an
                          //  experience slot
  space?: { mode?: "stereo" | "3d";
            pan; distance; rolloff; muffle };
}                         // space: target slots only
interface SoundPreset {
  source: { synth; params?; asset? };
  filter?; delay?; reverb?; lfo?;  // each enabled?
}

Plugin API: a synth, like every register*

const SPEC = {
  label: "FM bell",
  params: {
    pitch: { kind: "number", min: 40, max: 8000,
             log: true, default: 880 },
    ratio: { kind: "number", min: .5, max: 8,
             default: 3.5 },
    index: { kind: "number", min: 0, max: 20 },
  },
};
api.registerSynth("fm", SPEC,
  ({ params, audio, time, destination }) => {
    // build nodes into destination
    return {
      stop: at => carrier.stop(at),
      pitched: [carrier.detune],  // LFO pitch
      tail: 1.2,                  // s after stop
    };
  });

Plugin API: playing and listening

// direct: a slot by name
api.sound.play("hum", { key: target.id });
api.sound.stop("hum");

// the hook is the slot's own `listens` –
// any plugin event reaches it:
api.events.emit("doorOpens");

// a rule (behaviour action, MINOR)
{ on: "tap", do: [{ sound: "creak" }] }

// the reader (FeedbackService)
feedback("tap")  // emits ui.tap + vibration
                 // from the slot's haptic
